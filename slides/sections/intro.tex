\section{Introduction}
\begin{frame}{Motivation}
    Problems with the current state of the Internet
    \begin{itemize}
        \item government control
        \item war causes distruptions
        \item political unrest
        \item free speech
    \end{itemize}
\end{frame}

\begin{frame}{Use Case}
    Provide resilient way to communicate during Internet shutdowns in case of disasters or political unrest (Gaza war, political control in Palestine)
\end{frame}

\begin{frame}{Solution Approaches}
    \begin{itemize}
        \item To solve consolidation of power, decentralization is key \pause
        \begin{itemize}
            \item Different network type than IP networks: use p2p networks! \pause
            \begin{itemize}
                \item Structured p2p networks: Kademlia, BitTorrent, more recently DLT: ETH, Bitcoin, Ripple XRP, etc \pause
                \item Unstructured p2p networks: FireChat, Twimight, Briar, bitchat, Amigo \pause
            \end{itemize}
            \item How to use p2p networks without Internet: use p2p protocols! \pause
            \begin{itemize}
                \item BLE \pause
                \item WiFi-Direct
                \item Others: Zigbee, LoRA and others
            \end{itemize}
        \end{itemize}
    \end{itemize}
\end{frame}

\begin{frame}{Problem Statement, Approach, Contribution}
    \textbf{Problem Statement:} How to provide resilient decentralized communication over unstructured p2p networks using efficient p2p communication protocols in the presence of network disruptions? \pause

    \textbf{Approach:} Collect empirical data on signal propagation stochastics, use data to parameterize simulations of network topologies, measure resilience w.r.t centralization metrics \pause

    \textbf{Contribution:} Provide the design space for resilient unstructured p2p networks
\end{frame}
