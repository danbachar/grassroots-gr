\section{Evaluation}
\begin{frame}{Model Insights}
    \begin{itemize}
        \item Gini much more indicative of resilience, as $\mathscr{S}-Score$ saturates very quickly, not indicative of sparsity
        \item Centralization correlates with an increase in peer degree, especially in higher ranges
        % \item Communication range 10m removed, as it skewed the scale of the results (Gini+$\mathscr{S}-Score$ 1.0)
        \item Increasing maximum allowed peer degree increases centralization
        \item In lower communication ranges, increasing the maximum node degree has almost no effect on centralization
    \end{itemize}
    \begin{figure}
        \centering
        \includegraphics[width=\textwidth]{figures/degree_vs_centralization_inter_cluster.png}
        \caption{Centralization vs. Max Peer Degree}
    \end{figure}
\end{frame}
\begin{frame}{Model Insights}
    \begin{itemize}
        \item Increasing the maximum peer degree beyond a threshold decreases latency for long-distance messages
        \item Short-distance messages are penalized by a high peer degree due to increased contention
        \item Long tail of short-distance indicates most direct links are short ranged
    \end{itemize}
    \begin{figure}
        \centering
        \includegraphics[width=0.75\textwidth]{figures/lat_degree_inter_cluster_90.png}
        \caption{Latency vs. Max Peer Degree at BLE 1M PHY Range}
    \end{figure}
\end{frame}
