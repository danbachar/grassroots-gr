\begin{frame}{Resilience Metrics: Six Criteria of Sparsity}
    \begin{enumerate}
        \item Robin Hood: Stealing from the rich and giving to the poor decreases sparsity
        \item \textbf{Scaling}: multiplying all values by a constant does not change sparsity
        \item Rising Tide: Adding a constant to all values decreases sparsity
        \item Cloning: Cloning the vector does not change sparsity
        \item Bill Gates: Adding a large value to the vector increases sparsity
        \item Babies: Adding a zero to the vector increases sparsity
    \end{enumerate}
\end{frame}
