\begin{frame}{Testbed}
    \begin{itemize}
        \item Model advertisement success probability vs. distance using empirical RSSI data
        \item Use number of advertisements received at $d_0=0$
        \item Probability of advertisement being received at distance $d$:
        \[
            P_{adv}(d) = \frac{\text{Number of received advertisements at } d}{\text{Total number of received advertisements at } d_0}
        \]
    \end{itemize}

\begin{frame}{Simulations}
    \begin{itemize}
        \item \textbf{What} Measure resilience of communication clusters established using empirical BLE signal stochastics
        \item \textbf{How} The ONE Simulator, $400m\times400m$ L-shaped field, divided to $100m \times 100m$ grid cells
    \end{itemize}
    \begin{minipage}{0.45\textwidth}
        \centering
        \includegraphics[width=\textwidth]{figures/run.jpg}
        \captionof{figure}{Simulation Area}
    \end{minipage}
    \hfill
    \begin{minipage}{0.55\textwidth}
        \centering
        \begin{itemize}
            \item 12 cluster cells, 6 hosts per cell, static movement model
            \item Epidemic Router, same amount of messages sent between each host pair
            \item 25 simulator runs consist of: network interface communication range, maximum peer degree, communication mode
        \end{itemize}
    \end{minipage}
\end{frame}
\begin{frame}{Simulations}
    \begin{itemize}
        \item Parameterize network interface bandwidth based on link distance using RSSI results from testbed
        \item Use log-normal path loss model~\cite{pathloss} to interpolate singal propagation decay between discrete distance points
        \item BLE 1M PHY~\cite{adaptive-phy}: Use maximum channel bandwidth of 1 Mbps at $\le 1m$ distance to deduce message transfer time
    \end{itemize}
